\section{The Prokeimenon and Epistle}

\speak{Deacon}{Let us attend!}

\speak{Reader}{\textit{Chants the refrain of the prokeimenon.}}

\speak{People}{\textit{Repeat the refrain.}}

\speak{Reader}{\textit{Chants the verse, repeating multiple times if there are multiple
verses.}}

\speak{People}{\textit{Repeat the refrain in response to each verse.}}

\speak{Reader}{\textit{Chants part of the refrain.}}

\speak{People}{\textit{Complete the refrain, more slowly than usual.}}

\speak{Deacon}{Wisdom!}

\speak{Reader}{\textit{Announces the epistle reading.}}

\rubric{It is customary for the people to sit during the Epistle reading.}

\speak{Deacon}{Let us attend!}

\speak{Reader}{\textit{Reads the appointed epistle reading.}}

\speak{Priest}{Peace be to thee that readest.}

\speak{Reader}{And to thy spirit.}

\speak{People}{Alleluia. (3x)}

\speak{Reader}{\textit{Chants the first Alleluia verse.}}

\speak{People}{Alleluia. (3x)}

\speak{Reader}{\textit{Chants the second Alleluia verse.}}

\speak{People}{Alleluia. (3x)}

\section{The Gospel}

\rubric{Everyone should stand during the reading of the Gospel.}

\speak{Priest}{Peace be to all.}

\speak{People}{And to thy spirit.}

\speak{Priest}{\textit{Announces the gospel reading.}}

\speak{People}{Glory to Thee, O Lord, glory to Thee.}

\speak{Priest}{\textit{Reads the appointed gospel reading.}}

\speak{People}{\cross~Glory to Thee, O Lord, glory to Thee.}
