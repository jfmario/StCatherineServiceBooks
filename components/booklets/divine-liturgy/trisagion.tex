\section{The Trisagion}

\speak{Deacon}{Let us pray to the Lord.}

\speak{People}{Lord, have mercy.}

\speak{Priest}{For holy art Thou, O our God, and unto Thee we ascribe \cross~glory, to the
Father and to the Son and to the Holy Spirit, now and ever\ldots}

\speak{Deacon}{\ldots and unto ages of ages.}

\speak{People}{Amen.}

\rubric{The Trisagion (thrice-holy hymn) is usually sung here. Sometimes, this is replaced
by an alternate hymn for baptism or the cross. This might be indicated in the bulletin.}

\subsect{The Trisagion (Tone 3)}

\speak{People}{\cross~Holy God, Holy Mighty, Holy Immortal, have mercy on us. (3x)

\cross~Glory to the Father and to the Son and to the Holy Spirit, now and ever and unto
ages of ages. Amen.

Holy Immortal, have mercy on us.}

\rubric{The appointed Epistle reader begins making their way to the center.}

\speak{Deacon}{With strength!}

\speak{People}{\cross~Holy God, Holy Mighty, Holy Immortal, have mercy on us.}
