\section{The Little Entrance}

\rubric{If it is not a Sunday, ``Who art risen from the dead'' is replaced by
``Who is wondrous in the saints.''}

\speak{Deacon}{Wisdom! Let us attend!}

\speak{Choir}{Come, let us worship and fall down before Christ. O Son of God,
\uline{Who art risen from the dead}, save us who sing unto Thee: Alleluia. Alleluia. Alleluia.}

\section{Hymns of the Day}

\rubric{A number of variable hymns are inserted here.}

\subsect{Hymn for St Catherine}

\rubric{Skipped on major feasts.}

\speak{People}{Let us praise the all-lauded bride of Christ, Catherine, divine guardian of
Sinai. She is our help and our defense, who by the power of the spirit, silenced brilliantly
the affectations of the impious. Crowned as a martyr, she seeketh for all the great mercy.}

\subsect{Ordinary Kontakion}

\rubric{This is frequently replaced based on liturgical cycles.}

\speak{People}{
  O undisputed intercessor of Christians, the mediatrix who are unrejected by the Creator,
  turn not away from the voice of our petitions, though we be sinners.
  Come to us with aid in time, who cry unto thee in faith, for thou art good.
  Hasten to us with intercessions, O Theotokos, who dost ever intercede
  for those who honor thee.}
